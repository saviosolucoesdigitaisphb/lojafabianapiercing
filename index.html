📅 Conectar ao Google Agenda

Siga estes passos uma única vez. Depois, todo agendamento cai sozinho no seu Google Agenda.

  1. Acesse script.google.com e faça login com a conta do Google que tem o calendário.
  2. Clique em Novo projeto.
  3. Apague o código que aparecer e cole o código abaixo (use o botão copiar).
  4. Clique em Implantar → Nova implantação.
  5. Em "Tipo", escolha App da Web.
  6. Em "Quem pode acessar", escolha Qualquer pessoa.
  7. Clique em Implantar e autorize o acesso (pode aparecer um aviso — clique em "Avançado" e depois "Acessar").
  8. Copie a URL do app da Web que aparecer e cole no campo do sistema.
function doPost(e) {
  try {
    var dados = JSON.parse(e.postData.contents);
    var cal = CalendarApp.getDefaultCalendar();

    // monta início e fim a partir de data + hora + duração
    var partesData = dados.data.split('-');   // YYYY-MM-DD
    var partesHora = dados.hora.split(':');    // HH:MM
    var inicio = new Date(
      partesData[0], partesData[1]-1, partesData[2],
      partesHora[0], partesHora[1], 0
    );
    var fim = new Date(inicio.getTime() + (dados.duracao||30)*60000);

    var titulo = '💎 ' + dados.servico + ' - ' + dados.cliente;
    var descricao =
      'Cliente: ' + dados.cliente + '\n' +
      'Telefone: ' + dados.fone + '\n' +
      'Serviço: ' + dados.servico + '\n' +
      'Duração: ' + dados.duracao + ' min\n' +
      'Origem: ' + dados.origem;

    var evento = cal.createEvent(titulo, inicio, fim, {description: descricao});
    // lembrete: 1 dia antes (por e-mail) e 60 min antes (popup)
    evento.addEmailReminder(24*60);
    evento.addPopupReminder(60);

    return ContentService.createTextOutput('OK');
  } catch (err) {
    return ContentService.createTextOutput('ERRO: ' + err);
  }
}
